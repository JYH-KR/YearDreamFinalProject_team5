CREATE TABLE user(
	id          VARCHAR(50)  NOT NULL  COMMENT '아이디'
	,pw         VARCHAR(100) NOT NULL  COMMENT '비밀번호'
	,name       VARCHAR(50)  NOT NULL  COMMENT '이름'
	,phone      VARCHAR(20)            COMMENT '전화번호'
	,email      VARCHAR(100)           COMMENT '이메일'
	,ins_dt     DATE                   COMMENT '등록일자'
	,ins_nm     VARCHAR(50)            COMMENT '등록자'
	,upt_dt     DATE                   COMMENT '수정일자'
	,upt_nm     VARCHAR(50)            COMMENT '수정자'
	,PRIMARY KEY(id)
) DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='회원';

CREATE TABLE safety_facilities(
	seq         INT          NOT NULL AUTO_INCREMENT  COMMENT '순번'
	,type       VARCHAR(10)  NOT NULL  COMMENT '시설타입 (S:가로등, C:CCTV)'
	,address    VARCHAR(100)           COMMENT '주소'
	,qty		INT					   COMMENT '설치개수'	
	,lat        DOUBLE       NOT NULL  COMMENT '좌표값(X)'
	,lng        DOUBLE       NOT NULL  COMMENT '좌표값(Y)'
	,base_date  DATE                   COMMENT '데이터 기준일자'
	,ins_dt     DATE                   COMMENT '등록일자'
	,ins_nm     VARCHAR(50)            COMMENT '등록자'
	,upt_dt     DATE                   COMMENT '수정일자'
	,upt_nm     VARCHAR(50)            COMMENT '수정자'
	,PRIMARY KEY(seq)
	,INDEX idx_type (type)
	,INDEX idx_latlng (lat, lng)
) DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='안전시설';
